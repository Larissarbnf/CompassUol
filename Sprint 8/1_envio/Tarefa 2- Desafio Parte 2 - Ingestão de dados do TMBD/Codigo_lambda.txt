def get_imdb_id(movie_id):
    imdb_id_url = f"https://api.themoviedb.org/3/movie/{movie_id}/external_ids"
    params = {"api_key": api_key}
    response = requests.get(imdb_id_url, params=params)

    if response.status_code == 200:
        data = response.json()
        return data.get("imdb_id")
    else:
        print(f"Falha ao obter IMDb ID para o filme {movie_id}. Código de status:", response.status_code)
        return None

def get_hbo_romance_movies():
    common_params = {
        "api_key": api_key,
        "include_adult": False,
        "include_video": False,
        "language": "en-US",
        "sort_by": "popularity.desc",
        "with_genres": "10749",  # Gênero de Romance
        "primary_release_year": 2023,
        "with_original_language": "en",
        "with_companies": "hbo"
    }

    hbo_romance_movies = []

    for page in range(1, 88):  # 87 páginas
        common_params["page"] = page

        discover_url = "https://api.themoviedb.org/3/discover/movie"
        discover_response = requests.get(discover_url, params=common_params)

        if discover_response.status_code == 200:
            discover_data = discover_response.json()

            for movie in discover_data.get("results", []):
                movie_id = movie.get("id")
                imdb_id = get_imdb_id(movie_id)
                movie["imdb_id"] = imdb_id

            hbo_romance_movies.extend(discover_data.get("results", []))
        else:
            print(f"Falha ao obter filmes de romance da HBO na página {page}. Código de status:", discover_response.status_code)

    return hbo_romance_movies

def ingest_romance_data(output_folder):
    hbo_romance_movies = get_hbo_romance_movies()

    # Dividir os filmes em lotes de 100
    batch_size = 100
    num_batches = math.ceil(len(hbo_romance_movies) / batch_size)
    
    for batch_num in range(num_batches):
        start_idx = batch_num * batch_size
        end_idx = (batch_num + 1) * batch_size
        batch_movies = hbo_romance_movies[start_idx:end_idx]

      
        df = pd.json_normalize(batch_movies)

        json_content = df.to_json(orient='records')
        output_key = f"{output_folder}output_batch_{batch_num + 1}.json"
        s3_client.put_object(Bucket=bucket_name, Key=output_key, Body=json_content)
        
        print(f"Arquivo {output_key} transferido com sucesso.")
